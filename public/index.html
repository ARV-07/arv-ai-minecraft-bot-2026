const express = require('express'); const path = require('path'); const http = require('http'); const { WebSocketServer } = require('ws'); function startServer({ port, host, botManager, memory, ai, commands }) { const app = express(); app.use(express.json()); app.use(express.static(path.join(__dirname, '../../public'))); const auth = (req, res, next) => { const username = process.env.DASHBOARD_USERNAME || 'admin'; const password = process.env.DASHBOARD_PASSWORD || 'change-me'; const header = req.headers.authorization || ''; const expected = 'Basic ' + Buffer.from(`${username}:${password}`).toString('base64'); if (header !== expected) { res.setHeader('WWW-Authenticate', 'Basic realm="ARV Bot Dashboard"'); return res.status(401).json({ error: 'Unauthorized' }); } next(); }; app.get('/api/status', auth, (req, res) => { res.json({ ok: true, data: botManager.getState(), ai: ai.state }); }); app.post('/api/command', auth, async (req, res) => { const { text } = req.body || {}; if (!text) return res.status(400).json({ error: 'Command text is required' }); try { const result = await commands.route(text); res.json(result); } catch (error) { res.status(500).json({ error: error.message }); } }); app.get('/api/logs', auth, (req, res) => { res.json({ ok: true, data: botManager.getLogs() }); }); app.get('/api/memory', auth, (req, res) => { res.json({ ok: true, data: memory.getTasks() }); }); const server = http.createServer(app); const wss = new WebSocketServer({ server }); wss.on('connection', (ws) => { ws.send(JSON.stringify({ type: 'connected', payload: { ok: true } })); const ping = setInterval(() => { if (ws.readyState === ws.OPEN) { ws.ping(); } }, 30000); ws.on('close', () => clearInterval(ping)); }); const broadcast = (data) => { const message = JSON.stringify(data); for (const client of wss.clients) { if (client.readyState === 1) client.send(message); } }; const sendStatus = () => { broadcast({ type: 'status', payload: { state: botManager.getState(), ai: ai.state } }); }; setInterval(sendStatus, 2000); return new Promise((resolve) => { server.listen(port, host, () => { console.log(`Dashboard listening on http://${host}:${port}`); resolve({ app, server, broadcast }); }); }); } module.exports = { startServer };
